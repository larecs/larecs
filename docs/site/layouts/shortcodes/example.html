{{- /* Renders a documentation example program from `examples/`. Usage: {{< example file="particles/movement.mojo" region="kernel" >}} {{< example file="particles/movement.mojo" full=true >}} `region` selects the lines between `# region NAME` and `# endregion NAME` and removes their common indentation. Without `region`, the whole file is shown. `full=true` wraps the whole program in a collapsed block with a link to its source. Region marker lines are never shown. A missing file or region fails the build. */ -}} {{- $file := .Get "file" -}} {{- $region := .Get "region" -}} {{- $full := .Get "full" -}} {{- $resource := resources.Get (printf "examples/%s" $file) -}} {{- if not $resource -}} {{- errorf "example: file %q not found under examples/ (%s)" $file .Position -}} {{- end -}} {{- $markerRE := `^\s*# (end)?region\b` -}} {{- $lines := split (strings.TrimRight "\n" $resource.Content) "\n" -}} {{- $selected := slice -}} {{- if $region -}} {{- $inside := false -}} {{- $found := false -}} {{- range $lines -}} {{- if findRE (printf `^\s*# endregion %s\s*$` $region) . -}} {{- $inside = false -}} {{- else if findRE (printf `^\s*# region %s\s*$` $region) . -}} {{- $inside = true -}} {{- $found = true -}} {{- else if and $inside (not (findRE $markerRE .)) -}} {{- $selected = $selected | append . -}} {{- end -}} {{- end -}} {{- if not $found -}} {{- errorf "example: region %q not found in %s (%s)" $region $file .Position -}} {{- end -}} {{- else -}} {{- range $lines -}} {{- if not (findRE $markerRE .) -}} {{- $selected = $selected | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Remove the indentation shared by all non-blank lines. */ -}} {{- $indent := 1000 -}} {{- range $selected -}} {{- if strings.TrimSpace . -}} {{- $width := len (replaceRE `^( *).*$` "$1" .) -}} {{- if lt $width $indent -}}{{- $indent = $width -}}{{- end -}} {{- end -}} {{- end -}} {{- $code := slice -}} {{- range $selected -}} {{- if ge (len .) $indent -}} {{- $code = $code | append (substr . $indent) -}} {{- else -}} {{- $code = $code | append "" -}} {{- end -}} {{- end -}} {{- $highlighted := partial "code-block.html" (strings.Trim (delimit $code "\n") "\n") -}} {{- if $full -}}
Complete program: examples/{{ $file }} {{ $highlighted }}
